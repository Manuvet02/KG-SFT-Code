CQ1. 	Where are the main sports facilities located in Trentino?

SELECT 
	(REPLACE(STR(?sport_facility), "http://localhost:8080/source/SportFacility/","") AS ?id) 
	?city ?address
WHERE {
   ?sport_facility a sft:sport_facility ;
 	sft:has_city ?city ;
  	sft:has_address ?address .
}

CQ5.	Which sports facilities regularly host a specific team or association?

SELECT 
	(REPLACE(STR(?sport_facility), "http://localhost:8080/source/SportFacility/","") AS ?id) 
	(REPLACE(STR(?association), "http://localhost:8080/source/Association/","") AS ?team)
WHERE {
   ?association a sft:association ;
     	sft:has_sport ?sport .

   {
    	SELECT ?sport (MIN(?sport_facility) AS ?sport_facility)
    	WHERE {
      	   ?sport_facility a sft:sport_facility ;
         	sft:has_sport ?sport .
    	}
    	GROUP BY ?sport
   }
}

CQ9.	Which public transport options are available to reach a specific sports facility?

	(REPLACE(STR(?sport_facility), "http://localhost:8080/source/SportFacility/","") AS ?id)

CQ11.	What is the average travel time by public transport to a facility?

SELECT 
	(REPLACE(STR(?sport_facility), "http://localhost:8080/source/SportFacility/","") AS ?id) 
	(AVG(?time) AS ?avg_time)
WHERE {
   ?sport_facility a sft:sport_facility ;
     	sft:has_meanOfTransport ?mean_of_transport .

   ?route a sft:route ;
     	sft:has_meanOfTransport ?mean_of_transport ;
     	sft:has_time ?time .
}
GROUP BY ?sport_facility

CQ12.	Are there bike paths connecting a sports facility to the rest of the city?

SELECT DISTINCT 
(REPLACE(STR(?sport_facility), "http://localhost:8080/source/SportFacility/","") AS ?idFacility) 
(REPLACE(STR(?path), "http://localhost:8080/source/Path/","") AS ?idPath)
WHERE
{ 
   ?sport_facility a sft:sport_facility ;
     	sft:has_meanOfTransport ?mean_of_transport .

   ?route a sft:route ;
     	sft:has_meanOfTransport ?mean_of_transport .

   ?path a sft:path ;
     	sft:has_route ?route ;
     	sft:has_type "bike line" .
}
